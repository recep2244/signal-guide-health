       ```typescript
       /**
        * Hook to record a new cardiac metric entry for a patient
        */
       export function useRecordCardiacMetric(patientId: string) {
         const queryClient = useQueryClient();
         return useMutation({
           mutationFn: async (payload: CardiacMetricInput) => {
             if (USE_MOCK) {
               // In mock mode, just return a stub — no network call
               return { status: 'success', data: { metric: { id: 'mock', patientId, ...payload } } };
             }
             const res = await fetch(`/api/patients/${patientId}/cardiac-metrics`, {
               method: 'POST',
               headers: { 'Content-Type': 'application/json' },
               body: JSON.stringify(payload),
             });
             if (!res.ok) {
               const err = await res.json().catch(() => ({}));
               throw new Error(err.message ?? 'Failed to save cardiac metric');
             }
             return res.json();
           },
           onSuccess: () => {
             queryClient.invalidateQueries({ queryKey: patientDataKeys.detail(patientId) });
           },
         });
       }
       ```
